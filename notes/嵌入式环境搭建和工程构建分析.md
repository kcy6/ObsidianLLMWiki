### Printf重定向
gcc
```c
int _write(int fd, char *pBuffer, int size)
{
	// 避免串口发送过程中的死循环,加入超时机制 
	const uint32_t timeout = 100000; // 超时周期,假设最多等待 100,000 次
	uint32_t timeout_counter = 0;
	
	for (int i = 0; i < size; i++) 
	{ 
		// 等待直到串口的数据寄存器空 
		while ((USART1->SR & 0x40) == 0) 
		{ 
			timeout_counter++; 
			if (timeout_counter >= timeout) 
			{ 
				// 如果超过超时限制,可以跳出并返回错误,或者做其他处理 
				return -1; // 返回错误 
			} 
		} 
		USART1->DR = (uint8_t) pBuffer[i]; // 写入数据寄存器,发送字符 
	} 
	return size; // 返回成功发送的字符数量 
}
```

**自动编译并且烧录脚本**，该脚本最后一行仍需根据自己设备修改
```bash
# 默认使用 Release preset，可通过参数指定：./run.sh Debug
PRESET=${1:-Release}
BUILD_DIR="build/${PRESET}"
OUTPUT_DIR="${BUILD_DIR}"    # 产物输出目录，可独立修改

# 删除该 preset 对应的构建目录（clean build）
rm -rf "${BUILD_DIR}"
# 使用 CMakePresets.json 进行配置
cmake --preset "$PRESET"

# 使用 preset 构建
cmake --build --preset "$PRESET"

# 自动获取项目名：优先从 .ioc 文件提取，其次从 CMakeLists.txt 提取
if [ -f *.ioc ]; then
    PROJECT_NAME=$(basename *.ioc .ioc)
else
    PROJECT_NAME=$(grep -oP 'set\(CMAKE_PROJECT_NAME\s+\K[^)]+' CMakeLists.txt || echo "project")
fi
elf_file="${BUILD_DIR}/${PROJECT_NAME}.elf"
bin_file="${OUTPUT_DIR}/${PROJECT_NAME}.bin"
hex_file="${OUTPUT_DIR}/${PROJECT_NAME}.hex"

# 检查 ELF 文件是否存在
if [ ! -f "$elf_file" ]; then
    echo "Error: ELF file $elf_file not found!"
    exit 1
fi

# 生成二进制文件和 HEX 文件
arm-none-eabi-objcopy -O binary "$elf_file" "$bin_file"
arm-none-eabi-objcopy -O ihex "$elf_file" "$hex_file"

# 检查产物是否生成成功
for f in "$bin_file" "$hex_file"; do
    if [ ! -f "$f" ]; then
        echo "Error: File $f not generated!"
        exit 1
    fi
done

# 打印文件大小信息
echo "ELF file size:"
arm-none-eabi-size "$elf_file"
echo "Binary file information:"
file "$bin_file"

echo "HEX file size:"
arm-none-eabi-size "$hex_file"

# 烧录
openocd -f interface/cmsis-dap.cfg -f target/stm32f4x.cfg -c "program $hex_file verify reset exit"
```

#!/bin/bash

# 定义颜色代码
COLOR_RED="\e[31m"
COLOR_GREEN="\e[32m"
COLOR_YELLOW="\e[33m"
COLOR_BLUE="\e[34m"
COLOR_BOLD_GREEN="\e[1;32m"

if [ $? -ne 0 ]; then
    echo -e "${COLOR_RED}构建失败！${COLOR_RESET}"
    exit 1
fi

# 获取当前工程所在的盘符
drive_letter=$(pwd | cut -d'/' -f2 | tr '[:lower:]' '[:upper:]')

# 修改 compile_commands.json 文件中的路径
sed -i "s|/c/|C:/|g; s|/d/|D:/|g; s|/e/|E:/|g; s|/f/|F:/|g; s|/g/|G:/|g; s|/h/|H:/|g; s|/i/|I:/|g; s|/j/|J:/|g; s|/k/|K:/|g; s|/l/|L:/|g; s|/m/|M:/|g; s|/n/|N:/|g; s|/o/|O:/|g; s|/p/|P:/|g; s|/q/|Q:/|g; s|/r/|R:/|g; s|/s/|S:/|g; s|/t/|T:/|g; s|/u/|U:/|g; s|/v/|V:/|g; s|/w/|W:/|g; s|/x/|X:/|g; s|/y/|Y:/|g; s|/z/|Z:/|g" compile_commands.json

# 获取 .ioc 文件的前缀
ioc_file=$(basename ../*.ioc .ioc)
elf_file="$ioc_file.elf"
bin_file="$ioc_file.bin"
hex_file="$ioc_file.hex"

# 检查 ELF 文件是否存在
if [ ! -f "$elf_file" ]; then
    echo -e "${COLOR_RED}错误：未找到 ELF 文件 $elf_file！${COLOR_RESET}"
    exit 1
fi

# 生成二进制文件和 HEX 文件
echo -e "${COLOR_BLUE}生成二进制文件...${COLOR_RESET}"
arm-none-eabi-objcopy -O binary "$elf_file" "$bin_file"
echo -e "${COLOR_BLUE}生成 HEX 文件...${COLOR_RESET}"
arm-none-eabi-objcopy -O ihex "$elf_file" "$hex_file"

# 检查二进制文件是否生成成功
if [ ! -f "$bin_file" ]; then
    echo -e "${COLOR_RED}错误：未生成二进制文件 $bin_file！${COLOR_RESET}"
    exit 1
fi

# 检查 HEX 文件是否生成成功
if [ ! -f "$hex_file" ]; then
    echo -e "${COLOR_RED}错误：未生成 HEX 文件 $hex_file！${COLOR_RESET}"
    exit 1
fi

# 打印内存使用情况（带彩色进度条）
print_memory_bar() {
    local used=$1
    local total=$2
    local label=$3
    
    # 转换为字节（输入单位KB）
    total_bytes=$((total * 1024))
    
    # 计算使用百分比
    if [ $total_bytes -eq 0 ]; then
        echo "错误：总大小不能为零！"
        return
    fi
    
    percent=$((used * 100 / total_bytes))
    ((percent > 100)) && percent=100

    # 根据百分比选择颜色
    if [ $percent -ge 90 ]; then
        color=$COLOR_RED
    elif [ $percent -ge 80 ]; then
        color=$COLOR_YELLOW
    else
        color=$COLOR_GREEN
    fi

    # 构建进度条
    bar_length=30
    filled=$((percent * bar_length / 100))
    empty=$((bar_length - filled))
    bar=$(printf "%${filled}s" | tr ' ' '█')
    space=$(printf "%${empty}s")

    # 转换使用量为KB并格式化输出
    used_kb=$(awk "BEGIN {printf \"%.2f\", $used/1024}")
    echo -e "${label}: ${color}${bar}${space}${COLOR_RESET} ${used_kb}KB/${total}KB (${percent}%)"
}

# 获取内存使用数据
echo -e "\n${COLOR_BOLD_GREEN}内存使用分析：${COLOR_RESET}"
size_info=$(arm-none-eabi-size "$elf_file" | tail -n1)
text=$(awk '{print $1}' <<< $size_info)
data=$(awk '{print $2}' <<< $size_info)
bss=$(awk '{print $3}' <<< $size_info)

# 计算各区域使用量（字节）
rom_used=$((text + data))      # Flash使用量 = text + data
ram_used=$((data + bss))       # RAM使用量 = data + bss
ccmram_used=$(arm-none-eabi-objdump -h $elf_file | awk '/.ccmram/ {print "0x"$3}' | xargs printf '%d') 

# 设置芯片内存参数（根据实际芯片修改以下数值）
FLASH_SIZE=128   # KB
RAM_SIZE=12     # KB
CCMRAM_SIZE=64    # KB

# 显示内存使用情况
print_memory_bar $rom_used $FLASH_SIZE "FLASH"
print_memory_bar $ram_used $RAM_SIZE "RAM"
[ $CCMRAM_SIZE -gt 0 ] && print_memory_bar $ccmram_used $CCMRAM_SIZE "CCMRAM"

# 计算编译耗时
end_time=$(date +%s)
duration=$((end_time - start_time))


# 打印编译成功信息
echo -e "\n${COLOR_RED}编译成功！！！${COLOR_RESET}"
echo -e "${COLOR_GREEN}编译耗时: ${COLOR_GREEN}${duration}秒${COLOR_RESET}"

##### 方案三，纯手搓
三类 Cmake 文件参考：
1. 公共的 _*.cmake_，这部分主要提供了编译器及其参数、处理器等信息的描述，是使用 Cmake管理工程时的共用部分；
2. 模块的 _CmakeLists.txt_，用来描述项目里会引用不同的模块（自己创建的或应用第三方的库）；
3. 工程的 _CmakeLists.txt_，该文件指定了具体的编译规则，并最终生成可执行文件；这个文件会引用 _1_、_2_ 两个文件；

##### VSCode+EIDE+GCC+Makefile+Clangd
**该工具链并没有使用make而是将Makefile手动填入EIDE中再来调用ARM-GCC编译**，CubeMX 生成了 Makefile，就是为了给 make 这个包工头看的。而 EIDE 这个插件，相当于自带了一个更高级、带图形界面的“现代化包工头”，所以 EIDE 就不需要看旧包工头的 Makefile 图纸了，它会自己去指挥 GCC 干活。

1. 使用 eide 新建一个 Cortex-M 空项目，需要选择项目的保存位置，切换工作空间，将加载新项目
2. 使用 CubeMX 新建一个Makefile项目，将生成目录设置到项目根目录下，即生成的整个工程是根目录下的一个文件夹(此为官方文档中的管理方式，具体的管理方式可自定义)
3. 点击eide工程栏的添加源文件夹，选择普通文件夹，将 `cubemx_source` 选中添加到项目中
4. 打开 `cubemx_source` 目录下的 `Makefile` 文件，将其中的 C_INCLUDE 内容复制到 包含目录中，注意复制时要去除 `-I` 前缀，并加上 `cubemx_source/` 路径前缀(根据管理方式不同这里自行判断头文件路径)
5. 打开 `cubemx_source` 目录下的 `Makefile` 文件，将其中的 C_DEFS 内容复制到 预处理器定义中注意复制的时候要去除掉 `-D` 前缀
6. 将 `cubemx_source` 目录下的.ld 文件的相对路径复制一下，然后填写到编译配置中的 链接脚本路径设置项中

可以在工程根目录下的 `.code-workspace`中添加以下配置，该设置使得 eide 能够自动搜索源文件夹内的 IncludePath，可以不用手动去设置 `包含目录` 配置项，插件会自动完成 IncludePath 的添加
```
	"settings": {
		"EIDE.SourceTree.AutoSearchIncludePath": true,
	}
```

# 名词说明&折腾的前置条件

#### 大体生态介绍

**GNU生态**(Linux上的)
它提供了大量基础工具，比如：
- GCC（编译器）
- GDB（调试器）
- Make（构建工具）
- Bash（Shell）
本质：  
“一整套开发工具 + 自由软件理念”

**LLVM生态**
它提供了大量编译相关基础组件，比如：
- Clang（编译器前端）
- LLDB（调试器）
- LLVM（优化器 + 代码生成后端）
- LLD（链接器）
- libc++（C++ 标准库实现）
- compiler-rt（运行时库，如 sanitizer 支持）

> [!notes]
> 
- **GCC 是 GNU 项目里的一个编译器**
- **MinGW项目理解为把 GNU 工具链（尤其是 GCC）移植到 Windows生成原生Windows程序**
- **LLVM是跨平台编译器基础设施，从一开始就支持 Windows / macOS / Linux**
- 
Windows 上的几种“类 Unix / GNU 方案”
1）MinGW-w64
   → 只解决“编译”（将linux程序转换生成 Windows 程序）
2）MSYS2
	=mingw64+pacman+msys
   → 工具链 + 包管理 + Shell（开发环境）
3）Cygwin
	=在 Windows 上模拟一个 Linux 运行环境
   → POSIX 层 + 工具链（运行环境）
- MinGW-w64：**只是编译器工具链**
- MSYS2：**包含 MinGW + 管理系统**
- Cygwin：**不用 MinGW，走自己的 POSIX 路**
参考文章：
[[MinGW-MinGW64-MSYS2]]

#### 搭建环境用到的各工具说明
**VSCode** 编辑器
**Cmake** 元构建系统，编写/读取CMakeLists.txt，自动生成 Makefile 或 ninja 文件
**Makefile** 构建规则文件，记录了文件之间的依赖关系和编译命令
**Make** 构建工具，读取 Makefile，决定哪些文件需要编译，并指挥 gcc 工作。
**Ninja** 类似 make，但速度更快，专注于极速构建，其读取的文件通常是build.ninja
**GCC** 编译器，将 .c/.cpp 源码翻译成机器能懂的二进制目标文件
**Clang** 编译器
**OpenOCD** GDB的服务器，与烧录器相连的部分
**Clangd** C/C++ 语言服务器，提供写代码时的智能提示、代码跳转、实时报错

#### clangd
https://clangd.llvm.org/
##### 简单介绍
clangd是一个它是一个让轻量级编辑器也能拥有 IDE 级别 C++ 开发体验的工具。简言之提供高亮，补全，代码导航...的工具

vscode中安装的clangd插件是一个提供可交换的前端，安装的clang+llvm-22.1.4里面的clangd.exe是一个后端(记得添加到环境变量)
成功配置后在vscode中CTRL+SHIFT+P调出命令面板使用clangd:Restart language server重建后即可开始体验其功能，
但是该工具依赖compile_commands.json工作，该文件它必须位于项目根目录，或通过 `--compile-commands-dir=` 指定其路径。该文件告诉 Clangd 每个源文件的具体编译参数（如包含路径、宏定义等）。

![[20260514-164344_嵌入式环境搭建和工程构建分析_7b1237d70c4d.png]]

##### 工作原理
clangd 不需要知道谁编译了代码，它只需要知道：
- **头文件搜索路径（`-I`）**
- **预定义宏（`-D`）**
- **编译器内置宏/路径**
对比说明：
- **clangd** 使用真正的 Clang 编译器前端解析代码，理解宏、模板等更精准
	- clangd也可以不和clang编译器一起使用，只要与之配合的编译器能提供其所需的路径即可实现相同工作效果
	- 代码层面	clangd 确实是 LLVM 项目的一部分，复用了 Clang 的前端代码来解析 C/C++
	- 使用层面	clangd 是一个独立的语言服务器，你完全可以用 ==GCC/MSVC/ARMCC 编译代码==，同时用 clangd 做 IDE 支持
- **Microsoft C/C++** 插件使用自家的 IntelliSense 引擎，配置方式不同
- clangd 通常**更快、内存占用更低**，尤其在大型代码库上，之所以快是因为其利用了编译器相当于借助了编译的过程探明了项目所以能给出准确快速的服务，而非智能感知(猜)。
- clangd 需要 ==compile_commands.json==（可通过 CMake 的 -DCMAKE_EXPORT_COMPILE_COMMANDS=Ture 生成）或 `.clangd` 配置文件来知道项目的编译参数

![[20260514-164344_嵌入式环境搭建和工程构建分析_830d8326ce40.png]]

##### .clangd文件
`.clangd` 是 Clangd 的配置文件，相当于给compile_commands.json文件打补丁，用于在项目中自定义 Clangd 的行为。这个文件通常位于项目的根目录下，Clangd 会在当前文件所在的所有父目录中搜索 `.clangd` 文件
`.clangd` 文件使用 **YAML**格式，可以包含以下几个主要部分：
- `CompileFlags`: 编译标志部分
- `Diagnostics`: 诊断设置部分
- `InlayHints`: 代码提示设置部分
- `Completion`: 代码补全设置部分
- `SemanticTokens`: 语义标记设置部分

添加下列配置
```
Diagnostics:
  UnusedIncludes: None
  MissingIncludes: None
```

###### ARMCC
相较于clangd配合GCC使用，配置起来更麻烦，因为ARMCC（ARM Compiler 5）有自己的命令行语法和语言扩展。
==armcc 的特殊挑战==

| 问题                | 说明                                                               |
| ----------------- | ---------------------------------------------------------------- |
| **命令行语法不同**       | armcc 用 `--c99`、`--cpu=Cortex-M3`、`-I path`（空格），和 GCC/Clang 风格不同 |
| **Keil MDK 项目格式** | uvprojx 不是标准构建系统，没有内置的 `compile_commands.json` 生成                |
| **内置函数/扩展**       | armcc 有 `__irq`、`__svc`、`__asm` 等特有语法和内置函数，Clang 可能不认识           |
| **系统头文件路径**       | armcc 的标准头文件位置和结构异于 GCC                                          |
